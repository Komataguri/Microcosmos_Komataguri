    <!-- Mobile viewport fix script -->
    <script>
        // Функція для динамічного розрахунку висоти елементів
        function calculateMobileLayout() {
            const navbar = document.querySelector('.navbar');
            const announcement = document.querySelector('.delivery-announcement');
            const heroSection = document.querySelector('.hero-section');
            
            if (window.innerWidth <= 991 && navbar && announcement && heroSection) {
                // Отримуємо реальну висоту навігаційного меню
                const navbarHeight = navbar.offsetHeight;
                
                // Встановлюємо позицію оголошення відносно реальної висоти меню
                announcement.style.top = navbarHeight + 'px';
                
                // Отримуємо висоту оголошення
                const announcementHeight = announcement.offsetHeight;
                
                // Встановлюємо відступ для hero секції
                const totalOffset = navbarHeight + announcementHeight;
                heroSection.style.paddingTop = (totalOffset + 20) + 'px';
                
                // Встановлюємо мінімальну висоту з урахуванням всіх елементів
                heroSection.style.minHeight = `calc(100vh - ${totalOffset}px)`;
            }
        }
        
        // Виконуємо розрахунки після повного завантаження DOM
        document.addEventListener('DOMContentLoaded', function() {
            // Невелика затримка для завершення рендерингу
            setTimeout(calculateMobileLayout, 100);
        });
        
        // Перерахунок при зміні розміру вікна
        window.addEventListener('resize', function() {
            setTimeout(calculateMobileLayout, 100);
        });
        
        // Перерахунок при зміні орієнтації
        window.addEventListener('orientationchange', function() {
            setTimeout(calculateMobileLayout, 300);
        });
        
        // Ініціалізація AOS анімацій
        document.addEventListener('DOMContentLoaded', function() {
            AOS.init({
                duration: 1000,
                once: true,
                offset: 50
            });
        });
    </script>
